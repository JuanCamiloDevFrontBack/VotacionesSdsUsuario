import { Component } from '@angular/core';
import { FormBuilder, FormGroup, Validators } from '@angular/forms';

@Component({
  selector: 'app-login',
  templateUrl: './login.component.html',
  styleUrls: ['./login.component.css']
})
export class LoginComponent {

  loginForm: FormGroup;

  constructor(private fb: FormBuilder) {
    this.loginForm = this.fb.group({
      email: ['', [Validators.required, Validators.email]],
      password: ['', Validators.required]
    });
  }

  onSubmit(): void {
    if (this.loginForm.valid) {
      console.log('Datos enviados:', this.loginForm.value);
      // Aquí iría la llamada al backend
    } else {
      this.loginForm.markAllAsTouched();
    }
  }
}

🧠 login.component.ts (Angular 21 standalone)

import { Component, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './login.component.html',
  styleUrl: './login.component.css'
})
export class LoginComponent {

  loading = signal(false);

  loginForm = this.fb.group({
    email: ['', [Validators.required, Validators.email]],
    password: ['', [Validators.required]]
  });

  constructor(private fb: FormBuilder) {}

  onSubmit(): void {
    if (this.loginForm.invalid) {
      this.loginForm.markAllAsTouched();
      return;
    }

    this.loading.set(true);

    // Simulación llamada API
    setTimeout(() => {
      console.log('Login data:', this.loginForm.value);
      this.loading.set(false);
    }, 1000);
  }
}

1️⃣ STACK Y DECISIONES ESTRATÉGICAS
🔹 Base de Datos Recomendada: PostgreSQL

Recomendación: PostgreSQL 16+
¿Por qué PostgreSQL para un sistema electoral?

    ACID real y robusto → crítico para integridad del voto.

    Soporte avanzado de transacciones y locking fino → control de concurrencia preciso.

    Row Level Security (RLS) → ideal para multi-provincia.

    Soporte JSONB → auditoría flexible.

    Índices avanzados (GIN, BRIN, partial indexes).

    Extensiones criptográficas (pgcrypto).

    Replicación nativa + alta disponibilidad.

    Excelente rendimiento en escritura concurrente.

📌 MySQL no ofrece el mismo nivel de sofisticación en RLS ni auditoría avanzada.
📌 MongoDB no es adecuado para voto crítico por consistencia eventual.

Conclusión: PostgreSQL es la opción institucional más sólida.

🧭 VISIÓN GENERAL DEL FLUJO

El proceso se divide en 6 fases:

1️⃣ Autenticación
2️⃣ Validación de elegibilidad
3️⃣ Generación de token anónimo
4️⃣ Emisión registrada
5️⃣ Registro del voto anónimo
6️⃣ Confirmación y auditoría
🟦 FASE 1 — AUTENTICACIÓN SEGURA
Objetivo:

Validar identidad sin exponer información innecesaria.
Flujo:

    Usuario ingresa credenciales.

    Spring Security valida.

    Se genera JWT corto (ej: 10 minutos).

    Se valida:

        Usuario activo

        Provincia correcta

        Rol = VOTER

🟦 FASE 2 — VALIDACIÓN DE ELEGIBILIDAD

Antes de mostrar candidatos:

Se ejecuta UseCase:

ValidarElegibilidadVoto

Verificaciones:

    Elección está ABIERTA

    Usuario pertenece a la provincia

    No existe registro en vote_emission

    Usuario activo

Consulta crítica:

SELECT 1
FROM vote_emission
WHERE election_id = ?
AND voter_id = ?
FOR UPDATE;

🔒 Esto previene condiciones de carrera.
🟦 FASE 3 — GENERACIÓN DE TOKEN ANÓNIMO

Aquí ocurre la separación clave.

Se genera:

anonymous_token = SHA256(
    voter_id +
    election_id +
    random_salt +
    timestamp
)

Este token:

    Se guarda en vote_emission

    No se usa en tabla vote

    Solo prueba que hubo emisión

⚠ El token no debe permitir reconstrucción.
🟦 FASE 4 — REGISTRO DE EMISIÓN (TRANSACCIÓN 1)

Transacción A:

BEGIN;

INSERT INTO vote_emission
(
 id,
 election_id,
 voter_id,
 province_id,
 emitted_at,
 anonymous_token
);

COMMIT;

Constraint:
UNIQUE(election_id, voter_id)

Si falla → intento doble voto.

⚠ Esta transacción no crea el voto.
🟦 FASE 5 — REGISTRO DEL VOTO ANÓNIMO (TRANSACCIÓN 2)

Aquí se ejecuta una segunda transacción independiente.

Esto es CLAVE para anonimato.

Transacción B:

BEGIN;

INSERT INTO vote
(
 id,
 election_id,
 candidate_id,
 province_id,
 integrity_hash,
 created_at
);

COMMIT;

Donde:

integrity_hash = SHA256(
    vote_id +
    candidate_id +
    election_id +
    server_secret
)

⚠ No se guarda voter_id.
⚠ No se guarda anonymous_token.
⚠ No existe relación posible.
🔐 ¿Por qué dos transacciones?

Si se hiciera en una sola:

Podría existir correlación temporal o en logs.

Separarlas reduce posibilidad de análisis forense interno.
🟦 FASE 6 — AUDITORÍA

Se registra:

audit_log

Pero:

    No incluye candidate_id.

    Solo incluye acción "VOTE_EMITTED".

    Hash encadenado.

Ejemplo:

hash_chain = SHA256(previous_hash + current_data)

🔐 PROTECCIÓN ANTI-CORRELACIÓN

Medidas adicionales:

    Retraso aleatorio entre emisión y voto (ej 50-200ms)

    Logs separados

    Servicios desacoplados (opcional microservicio)

    Pool de inserción para votos (batch async opcional)

🧮 FASE DE CIERRE Y ESCRUTINIO

Cuando election.status = CERRADA:

    Se bloquean inserts.

    Se calcula:

SELECT candidate_id, COUNT(*)
FROM vote
WHERE election_id = ?
GROUP BY candidate_id;

result_hash = SHA256(
   total_votes +
   sorted_candidate_results +
   server_secret
)

    Se guarda en election_result.
